! =====================================================================
!  ACCESS-2-MainBldg - Clean configuration
!  School Network Project | Catalyst 2960 (Packet Tracer)
!  Rooms: Student Lab, Guests Area
!  Placeholders to replace before pasting: <ADMIN_SECRET>, <NTP_KEY>
! =====================================================================
enable
configure terminal
!
! ---------- 1. BASIC SETTINGS ----------
hostname ACCESS-2-MainBldg
service password-encryption
service timestamps log datetime msec
clock timezone PHT 8 0
username admin privilege 15 secret <ADMIN_SECRET>
ip domain-name schoolnetwork.local
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
!
! ---------- 2. VTP (client) ----------
vtp domain SCHOOL_NET
vtp mode client
!
! ---------- 3. SPANNING TREE ----------
spanning-tree mode rapid-pvst
spanning-tree portfast bpduguard default
spanning-tree vlan 10,20,30 priority 16384
spanning-tree vlan 40,50,60 priority 20480
!
! ---------- 4. UPLINK TRUNKS TO CORES ----------
interface range GigabitEthernet0/1 - 2
 switchport mode trunk
 switchport trunk native vlan 111
 switchport trunk allowed vlan 10,20,30,40,50,60
 switchport nonegotiate
exit
interface GigabitEthernet0/1
 description TRUNK_TO_CORE_SW1
interface GigabitEthernet0/2
 description TRUNK_TO_CORE_SW2
exit
!
! ---------- 5. STUDENT LAB (VLAN 30 / 60) ----------
interface FastEthernet0/1
 description Student-Lab-Camera
 switchport mode access
 switchport access vlan 60
 spanning-tree portfast
interface FastEthernet0/5
 description Lab-PC1
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
interface FastEthernet0/6
 description Lab-PC2
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
interface FastEthernet0/7
 description Lab-PC3
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
interface FastEthernet0/8
 description Lab-PC4
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
interface FastEthernet0/9
 description Lab-PC5
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
exit
!
! ---------- 6. GUESTS AREA (VLAN 40 / 60) ----------
interface FastEthernet0/2
 description Guest-Area-Camera
 switchport mode access
 switchport access vlan 60
 spanning-tree portfast
interface FastEthernet0/15
 description PC-Guest1
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
interface FastEthernet0/16
 description Printer-Guest
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
exit
!
! ---------- 7. LIGHTWEIGHT APs (FlexConnect trunks) ----------
interface FastEthernet0/3
 description LAP-StudentLab_(FlexConnect)
 switchport mode trunk
 switchport trunk native vlan 10
 switchport trunk allowed vlan 10,20,30,40
 switchport nonegotiate
 spanning-tree portfast trunk
 spanning-tree bpduguard enable
interface FastEthernet0/4
 description LAP-Guests_(FlexConnect)
 switchport mode trunk
 switchport trunk native vlan 10
 switchport trunk allowed vlan 10,20,30,40
 switchport nonegotiate
 spanning-tree portfast trunk
 spanning-tree bpduguard enable
exit
!
! ---------- 8. UNUSED PORTS (parked + shut) ----------
interface range FastEthernet0/10 - 14 , FastEthernet0/17 - 24
 description UNUSED_PARKED
 switchport mode access
 switchport access vlan 999
 shutdown
exit
!
! ---------- 9. MANAGEMENT IP ----------
interface Vlan1
 no ip address
 shutdown
exit
interface Vlan10
 description MANAGEMENT
 ip address 10.1.10.12 255.255.255.0
 no shutdown
exit
ip default-gateway 10.1.10.1
!
! ---------- 10. NTP ----------
ntp authentication-key 1 md5 <NTP_KEY>
ntp authenticate
ntp trusted-key 1
ntp server 10.1.50.10 key 1
!
! ---------- 11. CONSOLE & SSH ACCESS ----------
ip access-list standard MGMT_ONLY
 permit 10.1.10.0 0.0.0.255
exit
line con 0
 login local
 exec-timeout 10 0
line vty 0 15
 access-class MGMT_ONLY in
 login local
 transport input ssh
 exec-timeout 10 0
exit
!
end
write memory
