! =====================================================================
!  ACCESS-3-Library - Clean configuration
!  School Network Project | Catalyst 2960 (Packet Tracer)
!  Rooms: Library Service, Library Reading Area
!  Placeholders to replace before pasting: <ADMIN_SECRET>, <NTP_KEY>
! =====================================================================
enable
configure terminal
!
! ---------- 1. BASIC SETTINGS ----------
hostname ACCESS-3-Library
service password-encryption
service timestamps log datetime msec
clock timezone PHT 8 0
username admin privilege 15 secret <ADMIN_SECRET>
ip domain-name schoolnetwork.local
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
!
! ---------- 2. VTP (client) ----------
vtp domain SCHOOL_NET
vtp mode client
!
! ---------- 3. SPANNING TREE ----------
spanning-tree mode rapid-pvst
spanning-tree portfast bpduguard default
spanning-tree vlan 40,50,60 priority 8192
spanning-tree vlan 10,20,30 priority 12288
!
! ---------- 4. UPLINK TRUNKS TO CORES ----------
interface range GigabitEthernet0/1 - 2
 switchport mode trunk
 switchport trunk native vlan 111
 switchport trunk allowed vlan 10,20,30,40,50,60
 switchport nonegotiate
exit
interface GigabitEthernet0/1
 description TRUNK_TO_CORE_SW1
interface GigabitEthernet0/2
 description TRUNK_TO_CORE_SW2
exit
!
! ---------- 5. LIBRARY SERVICE (VLAN 50 / 60) ----------
interface FastEthernet0/1
 description Printer-Library
 switchport mode access
 switchport access vlan 50
 spanning-tree portfast
interface FastEthernet0/5
 description PC-Library-Service1
 switchport mode access
 switchport access vlan 50
 switchport port-security
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
interface FastEthernet0/14
 description PC-Library-Service2
 switchport mode access
 switchport access vlan 50
 spanning-tree portfast
interface FastEthernet0/9
 description Library-Service-Camera
 switchport mode access
 switchport access vlan 60
 spanning-tree portfast
exit
!
! ---------- 6. LIBRARY READING AREA (VLAN 60) ----------
interface FastEthernet0/8
 description Reading-Area-Camera
 switchport mode access
 switchport access vlan 60
 spanning-tree portfast
exit
!
! ---------- 7. LIGHTWEIGHT AP (FlexConnect trunk) ----------
interface FastEthernet0/6
 description LAP-StudentLibrary_(FlexConnect)
 switchport mode trunk
 switchport trunk native vlan 10
 switchport trunk allowed vlan 10,20,30,40
 switchport nonegotiate
 spanning-tree portfast trunk
 spanning-tree bpduguard enable
exit
!
! ---------- 8. UNUSED PORTS (parked + shut) ----------
interface range FastEthernet0/2 - 4 , FastEthernet0/7 , FastEthernet0/10 - 13 , FastEthernet0/15 - 24
 description UNUSED_PARKED
 switchport mode access
 switchport access vlan 999
 shutdown
exit
!
! ---------- 9. MANAGEMENT IP ----------
interface Vlan1
 no ip address
 shutdown
exit
interface Vlan10
 description MANAGEMENT
 ip address 10.1.10.13 255.255.255.0
 no shutdown
exit
ip default-gateway 10.1.10.1
!
! ---------- 10. NTP ----------
ntp authentication-key 1 md5 <NTP_KEY>
ntp authenticate
ntp trusted-key 1
ntp server 10.1.50.10 key 1
!
! ---------- 11. CONSOLE & SSH ACCESS ----------
ip access-list standard MGMT_ONLY
 permit 10.1.10.0 0.0.0.255
exit
line con 0
 login local
 exec-timeout 10 0
line vty 0 15
 access-class MGMT_ONLY in
 login local
 transport input ssh
 exec-timeout 10 0
exit
!
end
write memory
